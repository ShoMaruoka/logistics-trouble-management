# インシデント管理ワークフロー改修_変更概要仕様書

## 1. 概要

### 1.1 改修目的
物流トラブル管理システムにおいて、新しいインシデント管理ワークフローを実装し、より効率的で追跡可能なインシデント処理プロセスを実現する。

### 1.2 改修範囲
- インシデントステータス管理の全面的な見直し
- 新しいワークフロープロセスの実装
- データベーススキーマの拡張
- フロントエンド・バックエンドAPIの修正
- 権限制御の強化

### 1.3 対象システム
- バックエンド: LogisticsTroubleManagement.API (.NET 8)
- フロントエンド: Next.js 15 React アプリケーション
- データベース: SQL Server

## 4. バックエンド変更仕様

### 4.1 既存システムとの統合戦略

#### 4.1.1 段階的ステータス移行アプローチ
現在のIncidentStatusは5段階（Open, InProgress, Resolved, Closed, Cancelled）で実装されており、新しい6段階ワークフローとの共存が必要です。

**統合戦略：**
- 既存ステータスは維持（後方互換性確保）
- 新しいWorkflowStatusを並行運用
- 段階的移行により既存機能への影響を最小化

#### 4.1.2 Incidentエンティティの段階的拡張

```csharp
{

    public DateTime? CompletionDate { get; private set; }
    public DateTime? PreventionProposalDate { get; private set; }
    public DateTime? EffectivenessConfirmationDate { get; private set; }
    
    // 作業内容（一部は既存実装済み）
    public string Cause { get; private set; }  // 既存実装済み
    public string ResponseContent { get; private set; }  // 新規追加
    public string PreventionMeasures { get; private set; }  // 既存実装済み
    public string EffectivenessStatus { get; private set; }  // 既存実装済み（enum型）
    public string EffectivenessComment { get; private set; }  // 既存実装済み
    
    // 新ワークフロー有効化メソッド
    public void EnableNewWorkflow()
    {
        // 既存ステータスから新ワークフローステータスにマッピング
        WorkflowStatus = Status switch
        {
            IncidentStatus.Open when string.IsNullOrEmpty(Category) => Domain.Enums.WorkflowStatus.Unclassified,
            IncidentStatus.Open => Domain.Enums.WorkflowStatus.Pending,
            IncidentStatus.InProgress => Domain.Enums.WorkflowStatus.InProgress,
            IncidentStatus.Resolved => Domain.Enums.WorkflowStatus.Completed,
            IncidentStatus.Closed => Domain.Enums.WorkflowStatus.EffectivenessConfirmed,
            _ => Domain.Enums.WorkflowStatus.Unclassified
        };
    }
    
    // 新しいワークフローメソッド
    public void Classify(int troubleTypeId, int damageTypeId, int warehouseId, 
        int shippingCompanyId, int totalShipments, int defectiveItems, 
        Priority priority, DateTime dueDate)
    {
        if (Status != IncidentStatus.Unclassified)
            throw new InvalidOperationException("未分類状態のインシデントのみ分類可能です");
            
        TroubleTypeId = troubleTypeId;
        DamageTypeId = damageTypeId;
        WarehouseId = warehouseId;
        ShippingCompanyId = shippingCompanyId;
        TotalShipments = totalShipments;
        DefectiveItems = defectiveItems;
        Priority = priority;
        DueDate = dueDate;
        Status = IncidentStatus.Pending;
        UpdatedAt = DateTime.UtcNow;
    }
    
    public void StartResponse(int warehouseStaffUserId)
    {
        if (Status != IncidentStatus.Pending)
            throw new InvalidOperationException("未対応状態のインシデントのみ対応開始可能です");
            
        ResponseStartDate = DateTime.UtcNow;
        Status = IncidentStatus.InProgress;
        UpdatedAt = DateTime.UtcNow;
    }
    
    public void AnalyzeCause(string cause)
    {
        if (Status != IncidentStatus.InProgress)
            throw new InvalidOperationException("対応中状態のインシデントのみ原因入力可能です");
            
        Cause = cause;
        CauseAnalysisDate = DateTime.UtcNow;
        UpdatedAt = DateTime.UtcNow;
    }
    
    public void CompleteResponse(string responseContent)
    {
        if (Status != IncidentStatus.InProgress)
            throw new InvalidOperationException("対応中状態のインシデントのみ対応完了可能です");
            
        ResponseContent = responseContent;
        CompletionDate = DateTime.UtcNow;
        Status = IncidentStatus.Completed;
        UpdatedAt = DateTime.UtcNow;
    }
    
    public void ProposePreventionMeasures(string preventionMeasures)
    {
        if (Status != IncidentStatus.Completed)
            throw new InvalidOperationException("対応済状態のインシデントのみ再発防止策提案可能です");
            
        PreventionMeasures = preventionMeasures;
        PreventionProposalDate = DateTime.UtcNow;
        Status = IncidentStatus.PreventionProposed;
        UpdatedAt = DateTime.UtcNow;
    }
    
    public void ConfirmEffectiveness(string effectivenessStatus, string effectivenessComment)
    {
        if (Status != IncidentStatus.PreventionProposed)
            throw new InvalidOperationException("再発防止策提案済状態のインシデントのみ有効性確認可能です");
            
        EffectivenessStatus = effectivenessStatus;
        EffectivenessComment = effectivenessComment;
        EffectivenessConfirmationDate = DateTime.UtcNow;
        Status = IncidentStatus.EffectivenessConfirmed;
        UpdatedAt = DateTime.UtcNow;
    }
}
```

```csharp
// LogisticsTroubleManagement.API/Controllers/IncidentsController.cs（拡張）
[ApiController]
[Route("api/[controller]")]
[Authorize]
public class IncidentsController : BaseController
{
    // 既存メソッドは維持...
    private readonly IWorkflowValidationService _workflowValidationService;
    
    // 新ワークフローエンドポイント
    [HttpPost("{id}/enable-workflow")]
    public async Task<IActionResult> EnableWorkflow(int id)
    {
        var incident = await _incidentRepository.GetByIdAsync(id);
        if (incident == null) return NotFound();
        
        incident.EnableNewWorkflow();
        await _unitOfWork.SaveChangesAsync();
        return Ok();
    }
    
    [HttpPost("{id}/classify")]
    [Authorize(Roles = "Incident Manager")]
    public async Task<IActionResult> ClassifyIncident(int id, [FromBody] ClassifyIncidentDto dto)
    {
        try
        {
            var incident = await _incidentRepository.GetByIdAsync(id);
            if (incident == null) return NotFound();
            
            // マスタデータ存在チェック
            var isValid = await _workflowValidationService.ValidateMasterDataReferencesAsync(
                dto.TroubleTypeId, dto.DamageTypeId, dto.WarehouseId, dto.ShippingCompanyId);
            if (!isValid)
                return BadRequest(new { Error = "指定されたマスタデータが存在しません。" });
                
            // 新ワークフローが有効でない場合は有効化
            if (incident.WorkflowStatus == null)
                incident.EnableNewWorkflow();
                
            incident.Classify(dto.TroubleTypeId, dto.DamageTypeId, dto.WarehouseId, 
                dto.ShippingCompanyId, dto.TotalShipments, dto.DefectiveItems, 
                dto.Priority, dto.DueDate);
                
            await _unitOfWork.SaveChangesAsync();
            return Ok();
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "インシデント分類中にエラーが発生しました: ID={Id}", id);
            return StatusCode(500, new { Error = "インシデントの分類中にエラーが発生しました。" });
        }
    }
    
    [HttpPost("start-response")]
    [Authorize(Roles = "Warehouse Staff")]
    public async Task<IActionResult> StartResponse(StartResponseDto dto)
    {
        var result = await _workflowService.StartResponseAsync(GetCurrentUserId(), dto);
        return Ok(new { Success = result });
    }
    
    [HttpPost("analyze-cause")]
    [Authorize(Roles = "Warehouse Staff")]

    {
        var result = await _workflowService.AnalyzeCauseAsync(GetCurrentUserId(), dto);
        return Ok(new { Success = result });
    }
    
    [HttpPost("complete-response")]
    [Authorize(Roles = "Warehouse Staff")]
    public async Task<IActionResult> CompleteResponse(CompleteResponseDto dto)
    {
        var result = await _workflowService.CompleteResponseAsync(GetCurrentUserId(), dto);
        return Ok(new { Success = result });
    }
    
    [HttpPost("propose-prevention")]
    [Authorize(Roles = "Warehouse Staff")]
    public async Task<IActionResult> ProposePreventionMeasures(ProposePreventionDto dto)
    {
        var result = await _workflowService.ProposePreventionMeasuresAsync(GetCurrentUserId(), dto);
```

```typescript
interface WorkflowActionPanelProps {
  incident: IncidentWithWorkflow;
  userRole?: string;
  onWorkflowAction?: (action: string, data: any) => void;
}

export function WorkflowActionPanel({ incident, userRole, onWorkflowAction }: WorkflowActionPanelProps) {
  const renderActionForStatus = () => {
    switch (incident.workflowStatus) {
      case 'Unclassified':
        if (userRole === 'Incident Manager') {
          return <ClassificationForm incident={incident} onSubmit={onWorkflowAction} />
        }
        break
        
      case 'Pending':
        if (user?.roleName === 'Warehouse Staff') {
          return <StartResponseButton incident={incident} onClick={onWorkflowAction} />
        }
        break
        
      case 'InProgress':
        if (user?.roleName === 'Warehouse Staff') {
          return (
            <div className="space-y-4">
              <AnalyzeCauseForm incident={incident} onSubmit={onWorkflowAction} />
              <CompleteResponseForm incident={incident} onSubmit={onWorkflowAction} />
            </div>
          )
        }
        break
        
      case 'Completed':
        if (user?.roleName === 'Warehouse Staff') {
          return <ProposePreventionForm incident={incident} onSubmit={onWorkflowAction} />
        }
        break
        
      case 'PreventionProposed':
        if (user?.roleName === 'Incident Manager') {
          return <ConfirmEffectivenessForm incident={incident} onSubmit={onWorkflowAction} />
        }
        break
        
      default:
        return null
    }
  }
  
  return (
    <div className="workflow-section">
      <h3>ワークフローアクション</h3>
      {renderWorkflowActions()}
    </div>
  )
}
```

```typescript
export function useWorkflowMasterData() {
  const [data, setData] = useState<WorkflowMasterData | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  
  useEffect(() => {

      try {
        setLoading(true);
        // 既存のuseMasterDataフックと統合
        const response = await apiClient.get('/api/master-data/workflow-options');
        setData(response.data);
        setError(null);
      } catch (err) {
        setError('マスタデータの取得に失敗しました');
        console.error('Error fetching workflow master data:', err);
      } finally {
        setLoading(false);
      }
    };

  }, []);
  
  return { data, loading, error };
}
```
